**Goal:** Livrer la v1.2 (spec `docs/superpowers/specs/2026-08-09-v1.2-design.md`) en une seule PR : sélection visuelle des pins + caméra, note de fiche agrandie, halo atténué, copie d'adresse par icône + toast, popup explicative portant le toggle FR/EN, panneau liste triable, et refonte admin (tri, modal unifié, modèle géographique atomique, responsive, thème).

**Architecture:** Tout le rendu marqueur partagé passe par `src/lib/cookie-marker.ts` (MapLibre + Leaflet) ; la sélection devient un état visuel piloté par `applyMarkerSelection` + CSS. La caméra est unifiée : chaque carte réagit à `selected` via un effet (plus de easeTo/panTo inline dans les handlers de clic). Le panneau liste et la popup vivent dans `CookieMap` (au-dessus des deux moteurs). L'admin garde ses server actions inchangées ; l'atomicité adresse/position est une contrainte d'interface + géocodage inverse Photon.

**Tech Stack:** Next.js 16 App Router (conventions du repo), React client components, MapLibre GL + Leaflet, Tailwind v4 avec tokens `var(--*)`, vitest + @testing-library/react.

```ts
// Markup partagé des marqueurs (MapLibre crée ses éléments DOM à la main,
// Leaflet passe par divIcon.html) : une seule source pour les deux cartes.
function escapeHtml(s: string): string {
  return s.replaceAll('&', '&amp;').replaceAll('"', '&quot;').replaceAll('<', '&lt;')
}

export function cookieMarkerHtml(name: string, slug: string): string {
  const safe = escapeHtml(name)
  // La pastille nom (spec v1.2 §1) n'est visible qu'à l'état sélectionné — pilotée
  // par data-selected/data-dimmed via applyMarkerSelection + CSS.
  return `<button class="cmtl-pin-cookie" data-slug="${escapeHtml(slug)}" aria-label="${safe}"><svg viewBox="0 0 300 300" aria-hidden="true"><use href="#cmtl-cookie-full"/></svg><span class="cmtl-pin-name" aria-hidden="true">${safe}</span></button>`
}

// État visuel de sélection (spec v1.2 §1, variante C) : le sélectionné grossit et
// montre sa pastille, les autres s'estompent. root = conteneur de la carte.
export function applyMarkerSelection(root: ParentNode, slug: string | null): void {
  for (const el of root.querySelectorAll<HTMLElement>('.cmtl-pin-cookie')) {
    if (slug === null) {
      delete el.dataset.selected
      delete el.dataset.dimmed
    } else if (el.dataset.slug === slug) {
      el.dataset.selected = 'true'
      delete el.dataset.dimmed
    } else {
      el.dataset.dimmed = 'true'
      delete el.dataset.selected
    }
  }
}
```

```tsx
// Échelle de cookies + note chiffrée (layout validé v1.2 : cookies dominants,
// chiffre en Gill Sans avec « /5 » discret). `row` = version compacte du panneau liste.
export function RatingCookies({ rating, variant = 'sheet' }: { rating: number; variant?: 'sheet' | 'row' }) {
  const px = variant === 'sheet' ? 27 : 13
  const label = `${String(rating).replace('.', ',')} / 5`
  const kinds = Array.from({ length: 5 }, (_, i) => {
    if (rating >= i + 1) return 'full' as const
    if (rating >= i + 0.5) return 'half' as const
    return 'empty' as const
  })
  return (
    <span role="img" aria-label={label} className={`inline-flex items-center ${variant === 'sheet' ? 'gap-3' : 'gap-1.5'}`}>
      <span aria-hidden className="inline-flex items-center gap-1">
        {kinds.map((kind, i) => (
          <svg key={i} width={px} height={px} viewBox="0 0 300 300" data-cookie={kind}>
            <use href={`#cmtl-cookie-${kind}`} />
          </svg>
        ))}
      </span>
      {variant === 'sheet' ? (
        <span aria-hidden className="font-display text-[30px] leading-none text-[color:var(--accent-ink)]">
          {String(rating).replace('.', ',')}
          <span className="text-[14px] text-[color:var(--text-muted)]"> /5</span>
        </span>
      ) : (
        <span aria-hidden className="font-display text-[12px] leading-none text-[color:var(--accent-ink)]">
          {String(rating).replace('.', ',')}
        </span>
      )}
    </span>
  )
}
```

```ts
// Tri des cookies pour le panneau liste (public) et la liste admin.
// Pur et sans dépendance : testable sans DOM ni carte.
export type SortKey = 'distance' | 'name' | 'rating'
export type SortDir = 'asc' | 'desc'
export type Origin = { lat: number; lng: number }

const EARTH_RADIUS_M = 6371000

export function distanceMeters(a: Origin, b: Origin): number {
  const rad = (d: number) => (d * Math.PI) / 180
  const dLat = rad(b.lat - a.lat)
  const dLng = rad(b.lng - a.lng)
  const h =
    Math.sin(dLat / 2) ** 2 + Math.cos(rad(a.lat)) * Math.cos(rad(b.lat)) * Math.sin(dLng / 2) ** 2
  return 2 * EARTH_RADIUS_M * Math.asin(Math.sqrt(h))
}

// « 350 m » sous le kilomètre (arrondi à 10 m), « 1,2 km » au-dessus (virgule
// française, « ,0 » omis).
export function formatDistance(m: number): string {
  if (m < 1000) return `${Math.round(m / 10) * 10} m`
  const km = (m / 1000).toFixed(1).replace('.0', '').replace('.', ',')
  return `${km} km`
}

const byName = new Intl.Collator('fr', { sensitivity: 'base' })

export function sortShops<T extends { name: string; rating: number; lat: number; lng: number }>(
  shops: readonly T[],
  key: SortKey,
  dir: SortDir,
  origin?: Origin | null
): T[] {
  const sign = dir === 'desc' ? -1 : 1
  const sorted = [...shops]
  if (key === 'distance' && origin) {
    sorted.sort((a, b) => sign * (distanceMeters(origin, a) - distanceMeters(origin, b)))
  } else if (key === 'rating') {
    // Ex æquo départagés par le nom pour un ordre stable et prévisible.
    sorted.sort((a, b) => sign * (a.rating - b.rating) || byName.compare(a.name, b.name))
  } else {
    sorted.sort((a, b) => (key === 'name' ? sign : 1) * byName.compare(a.name, b.name))
  }
  return sorted
}
```

```ts
// Géocodage inverse (spec v1.2 §8) : l'adresse est TOUJOURS dérivée des
// coordonnées quand elle n'est pas fournie par la recherche — jamais saisie.
export async function reverseGeocode(lat: number, lng: number, fetchImpl: typeof fetch = fetch): Promise<string | null> {
  const url = new URL('https://photon.komoot.io/reverse')
  url.searchParams.set('lat', String(lat))
  url.searchParams.set('lon', String(lng))
  url.searchParams.set('lang', 'fr')
  try {
    const res = await fetchImpl(url.toString(), { headers: { 'User-Agent': 'cookies-mtl (personal project)' } })
    if (!res.ok) return null
    const data = (await res.json()) as { features?: PhotonFeature[] }
    const p = data.features?.[0]?.properties
    if (!p) return null
    const address = [p.housenumber && p.street ? `${p.housenumber} ${p.street}` : p.street, p.city]
      .filter(Boolean)
      .join(', ')
    return address || null
  } catch {
    return null
  }
}
```

```ts
export async function GET(request: Request) {
  if (!(await isAdmin())) return Response.json({ error: 'unauthorized' }, { status: 401 })
  const params = new URL(request.url).searchParams
  const lat = Number(params.get('lat'))
  const lng = Number(params.get('lng'))
  if (Number.isFinite(lat) && Number.isFinite(lng) && params.has('lat')) {
    return Response.json({ address: await reverseGeocode(lat, lng) })
```

```ts
  // Spec v1.2 §8 : jamais de saisie d'adresse. Coordonnées sans adresse
  // (lien Google, manuel, drag du pin) → géocodage inverse Photon via /api/places.
  useEffect(() => {
    if (!draft || draft.address !== '') return
    const { lat, lng } = draft
    let stale = false
    fetch(`/api/places?lat=${lat}&lng=${lng}`)
      .then((r) => (r.ok ? r.json() : null))

        setDraft((d) => (d && d.lat === lat && d.lng === lng ? { ...d, address: data.address } : d))

      .catch(() => {})
    return () => {
      stale = true
    }
  }, [draft?.lat, draft?.lng, draft?.address])
```
